\begin{algorithm}
\caption{Gradient Descent}
\begin{algorithmic}
\PROCEDURE{GradientDescent}{$f, \theta_0, \eta$}
    \STATE $\theta \gets \theta_0$
    \WHILE{not converged}
        \STATE $\theta \gets \theta - \eta \nabla f(\theta)$
    \ENDWHILE
    \RETURN $\theta$
\ENDPROCEDURE
\end{algorithmic}
\end{algorithm}
